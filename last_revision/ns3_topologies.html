01

Simple Point-to-Point (2 Nodes)

The most basic topology — first.cc. Two nodes connected by a direct P2P link.

// TOPOLOGY DIAGRAM
Nodes
2
Link Type
Point-to-Point
Network
10.1.1.0/24
Script
first.cc
Key Points:
  • DataRate is a device attribute → set via SetDeviceAttribute
  • Delay is a channel attribute → set via SetChannelAttribute
  • Server starts at t=1s, client at t=2s (always)
  • No need for Simulator::Stop() — queue empties naturally
// COMPLETE CODE — first.cc pattern
// ── INCLUDES ─────────────────────────────
#include "ns3/core-module.h"
#include "ns3/network-module.h"
#include "ns3/internet-module.h"
#include "ns3/point-to-point-module.h"
#include "ns3/applications-module.h"

using namespace ns3;

int main(int argc, char* argv[]) {

  // ── STEP 1: Create 2 nodes ────────────
  NodeContainer nodes;
  nodes.Create(2);

  // ── STEP 2: Configure P2P link ────────
  PointToPointHelper p2p;
  p2p.SetDeviceAttribute(
    "DataRate", StringValue("5Mbps"));
  p2p.SetChannelAttribute(
    "Delay", StringValue("2ms"));

  // ── STEP 3: Install on nodes ──────────
  NetDeviceContainer devices;
  devices = p2p.Install(nodes);

  // ── STEP 4: Install TCP/IP stack ──────
  InternetStackHelper stack;
  stack.Install(nodes);

  // ── STEP 5: Assign IP addresses ───────
  Ipv4AddressHelper address;
  address.SetBase(
    "10.1.1.0", "255.255.255.0");
  Ipv4InterfaceContainer ifaces =
    address.Assign(devices);
  // n0 → 10.1.1.1  |  n1 → 10.1.1.2

  // ── STEP 6: Install server on n1 ─────
  UdpEchoServerHelper server(9);
  ApplicationContainer sApps =
    server.Install(nodes.Get(1));
  sApps.Start(Seconds(1.0));
  sApps.Stop(Seconds(10.0));

  // ── STEP 7: Install client on n0 ─────
  UdpEchoClientHelper client(
    ifaces.GetAddress(1), 9);
  client.SetAttribute(
    "MaxPackets", UintegerValue(1));
  client.SetAttribute(
    "Interval", TimeValue(Seconds(1.0)));
  client.SetAttribute(
    "PacketSize", UintegerValue(1024));
  ApplicationContainer cApps =
    client.Install(nodes.Get(0));
  cApps.Start(Seconds(2.0));
  cApps.Stop(Seconds(10.0));

  // ── STEP 8: PCAP + Run ────────────────
  p2p.EnablePcapAll("first");
  Simulator::Run();
  Simulator::Destroy();
  return 0;
}
02

Linear Chain (4 Nodes)

Nodes connected end-to-end: n0 — n1 — n2 — n3. Good for studying multi-hop routing.

// TOPOLOGY DIAGRAM
n0 ——[10.1.1.0]—— n1 ——[10.1.2.0]—— n2 ——[10.1.3.0]—— n3 │ │ Client Server
Key Points:
  • Each P2P link is a separate subnet (10.1.1.0, 10.1.2.0, 10.1.3.0)
  • Must call PopulateRoutingTables() — packet crosses multiple subnets
  • Middle nodes (n1, n2) act as routers
  • Use loop to build N-node chains easily
// LINEAR CHAIN — 4 nodes
#include "ns3/core-module.h"
#include "ns3/network-module.h"
#include "ns3/internet-module.h"
#include "ns3/point-to-point-module.h"
#include "ns3/applications-module.h"
using namespace ns3;

int main(int argc, char* argv[]) {

  // ── 4 nodes in a chain ────────────────
  NodeContainer nodes;
  nodes.Create(4);     // n0, n1, n2, n3

  PointToPointHelper p2p;
  p2p.SetDeviceAttribute(
    "DataRate", StringValue("10Mbps"));
  p2p.SetChannelAttribute(
    "Delay", StringValue("1ms"));

  // ── 3 separate P2P links ─────────────
  NetDeviceContainer d01, d12, d23;
  d01 = p2p.Install(
    nodes.Get(0), nodes.Get(1));  // n0—n1
  d12 = p2p.Install(
    nodes.Get(1), nodes.Get(2));  // n1—n2
  d23 = p2p.Install(
    nodes.Get(2), nodes.Get(3));  // n2—n3

  // ── Install stack on ALL nodes ────────
  InternetStackHelper stack;
  stack.Install(nodes);

  // ── Each link gets its own subnet ─────
  Ipv4AddressHelper addr;

  addr.SetBase("10.1.1.0", "255.255.255.0");
  Ipv4InterfaceContainer i01 =
    addr.Assign(d01);

  addr.SetBase("10.1.2.0", "255.255.255.0");
  Ipv4InterfaceContainer i12 =
    addr.Assign(d12);

  addr.SetBase("10.1.3.0", "255.255.255.0");
  Ipv4InterfaceContainer i23 =
    addr.Assign(d23);

  // ── Routing (REQUIRED! multi-subnet) ──
  Ipv4GlobalRoutingHelper::
    PopulateRoutingTables();

  // ── Server on n3, client on n0 ────────
  UdpEchoServerHelper server(9);
  ApplicationContainer sApps =
    server.Install(nodes.Get(3));
  sApps.Start(Seconds(1.0));
  sApps.Stop(Seconds(10.0));

  // Client connects to n3's IP on link i23
  UdpEchoClientHelper client(
    i23.GetAddress(1), 9);
  client.SetAttribute(
    "MaxPackets", UintegerValue(3));
  client.SetAttribute(
    "PacketSize", UintegerValue(512));
  ApplicationContainer cApps =
    client.Install(nodes.Get(0));
  cApps.Start(Seconds(2.0));
  cApps.Stop(Seconds(10.0));

  p2p.EnablePcapAll("chain");
  Simulator::Run();
  Simulator::Destroy();
  return 0;
}
03

P2P + CSMA Bus Network

second.cc pattern — a P2P backbone with a bus (Ethernet/CSMA) LAN attached.

// TOPOLOGY DIAGRAM
n0 ——[P2P 10.1.1.0]—— n1 ===CSMA BUS 10.1.2.0=== n2 n3 │ │ │ │ P2P shared bystander Server
Key Points:
  • n1 is shared — added to both p2pNodes and csmaNodes. Install stack only once!
  • CSMA DataRate → channel attribute (not device)
  • ARP will appear in CSMA PCAP — Ethernet needs MAC resolution
  • Last CSMA node = csmaNodes.Get(nCsma)
  • Promiscuous PCAP → 3rd arg = true
// P2P + CSMA — second.cc pattern
#include "ns3/core-module.h"
#include "ns3/network-module.h"
#include "ns3/internet-module.h"
#include "ns3/point-to-point-module.h"
#include "ns3/csma-module.h"
#include "ns3/applications-module.h"
using namespace ns3;

int main(int argc, char* argv[]) {

  uint32_t nCsma = 3; // extra CSMA nodes

  // ── P2P nodes ─────────────────────────
  NodeContainer p2pNodes;
  p2pNodes.Create(2);  // n0, n1

  // ── CSMA nodes (n1 is shared!) ────────
  NodeContainer csmaNodes;
  csmaNodes.Add(p2pNodes.Get(1));  // n1
  csmaNodes.Create(nCsma);          // n2,n3

  // ── P2P link setup ────────────────────
  PointToPointHelper p2p;
  p2p.SetDeviceAttribute(
    "DataRate", StringValue("5Mbps"));
  p2p.SetChannelAttribute(
    "Delay", StringValue("2ms"));
  NetDeviceContainer p2pDevs =
    p2p.Install(p2pNodes);

  // ── CSMA link setup ───────────────────
  // NOTE: DataRate on CHANNEL (not device!)
  CsmaHelper csma;
  csma.SetChannelAttribute(
    "DataRate", StringValue("100Mbps"));
  csma.SetChannelAttribute(
    "Delay",
    TimeValue(NanoSeconds(6560)));
  NetDeviceContainer csmaDevs =
    csma.Install(csmaNodes);

  // ── Internet stack ────────────────────
  InternetStackHelper stack;
  stack.Install(p2pNodes.Get(0));  // n0
  stack.Install(csmaNodes);        // n1,n2,n3
  // n1 is in csmaNodes → don't Install twice

  // ── IP addresses ─────────────────────
  Ipv4AddressHelper addr;
  addr.SetBase("10.1.1.0", "255.255.255.0");
  Ipv4InterfaceContainer p2pIf =
    addr.Assign(p2pDevs);

  addr.SetBase("10.1.2.0", "255.255.255.0");
  Ipv4InterfaceContainer csmaIf =
    addr.Assign(csmaDevs);

  // ── Global routing (REQUIRED!) ────────
  Ipv4GlobalRoutingHelper::
    PopulateRoutingTables();

  // ── Server on last CSMA node ──────────
  UdpEchoServerHelper server(9);
  ApplicationContainer sApps =
    server.Install(csmaNodes.Get(nCsma));
  sApps.Start(Seconds(1.0));
  sApps.Stop(Seconds(10.0));

  // ── Client on n0 (P2P node) ───────────
  UdpEchoClientHelper client(
    csmaIf.GetAddress(nCsma), 9);
  client.SetAttribute(
    "MaxPackets", UintegerValue(1));
  client.SetAttribute(
    "PacketSize", UintegerValue(1024));
  ApplicationContainer cApps =
    client.Install(p2pNodes.Get(0));
  cApps.Start(Seconds(2.0));
  cApps.Stop(Seconds(10.0));

  // ── Tracing ───────────────────────────
  p2p.EnablePcapAll("second");
  // Promiscuous sniff on CSMA node index 1
  csma.EnablePcap(
    "second", csmaDevs.Get(1), true);

  Simulator::Run();
  Simulator::Destroy();
  return 0;
}
04

Star Topology (Hub + 4 Spokes)

Central hub node connected to multiple leaf nodes via P2P links.

// TOPOLOGY DIAGRAM
n3 | n1 — HUB(n0) — n2 | n4 n5 (each link = separate /24 subnet)
Key Points:
  • Hub node n0 gets one interface per link — N spokes = N devices on hub
  • Each P2P link = separate subnet. Hub needs stack installed once.
  • PopulateRoutingTables() required — all traffic routes via hub
  • Use loop for many spokes: p2p.Install(hub, nodes.Get(i))
// STAR — hub n0, leaves n1..n4
#include "ns3/core-module.h"
#include "ns3/network-module.h"
#include "ns3/internet-module.h"
#include "ns3/point-to-point-module.h"
#include "ns3/applications-module.h"
using namespace ns3;

int main(int argc, char* argv[]) {

  uint32_t nLeaves = 4;

  // ── Hub node + leaf nodes ─────────────
  Ptr<Node> hub = CreateObject<Node>();
  NodeContainer leaves;
  leaves.Create(nLeaves);

  PointToPointHelper p2p;
  p2p.SetDeviceAttribute(
    "DataRate", StringValue("10Mbps"));
  p2p.SetChannelAttribute(
    "Delay", StringValue("1ms"));

  // ── Install stacks ────────────────────
  InternetStackHelper stack;
  stack.Install(hub);
  stack.Install(leaves);

  // ── Connect hub to each leaf (loop) ───
  Ipv4AddressHelper addr;
  std::vector<Ipv4InterfaceContainer> ifaces;

  for (uint32_t i = 0; i < nLeaves; i++) {

    NodeContainer pair(hub, leaves.Get(i));
    NetDeviceContainer devs =
      p2p.Install(pair);

    // Each spoke gets a /24 subnet
    std::ostringstream base;
    base << "10.1." << (i+1) << ".0";
    addr.SetBase(
      base.str().c_str(),
      "255.255.255.0");
    ifaces.push_back(
      addr.Assign(devs));
  }

  // ── Routing ───────────────────────────
  Ipv4GlobalRoutingHelper::
    PopulateRoutingTables();

  // ── Server on leaf 3, client on leaf 0
  UdpEchoServerHelper server(9);
  ApplicationContainer sApps =
    server.Install(leaves.Get(3));
  sApps.Start(Seconds(1.0));
  sApps.Stop(Seconds(10.0));

  // ifaces[3].GetAddress(1) = leaf3's IP
  UdpEchoClientHelper client(
    ifaces[3].GetAddress(1), 9);
  client.SetAttribute(
    "MaxPackets", UintegerValue(1));
  client.SetAttribute(
    "PacketSize", UintegerValue(512));
  ApplicationContainer cApps =
    client.Install(leaves.Get(0));
  cApps.Start(Seconds(2.0));
  cApps.Stop(Seconds(10.0));

  Simulator::Run();
  Simulator::Destroy();
  return 0;
}
05

P2P + CSMA + WiFi (third.cc)

Full mixed topology — WiFi STAs → AP (P2P backbone) → CSMA LAN with server.

// TOPOLOGY DIAGRAM
WiFi(10.1.3.0) CSMA(10.1.2.0) STA(n4,n5,n6) ← → AP(n0) —P2P(10.1.1.0)— n1 ===BUS=== n2 n3 Server ↑ shared node
⚠ Critical for this topology:
  • WiFi beacons are self-sustaining → must call Simulator::Stop(Seconds(10))
  • n0 is the AP — uses ConstantPositionMobilityModel
  • STAs use RandomWalk2dMobilityModel
  • 3 networks → 3 SetBase() calls
  • ActiveProbing = false on STAs (listen for beacons, don't send probes)
// P2P+CSMA+WiFi — third.cc pattern
#include "ns3/core-module.h"
#include "ns3/network-module.h"
#include "ns3/internet-module.h"
#include "ns3/point-to-point-module.h"
#include "ns3/csma-module.h"
#include "ns3/wifi-module.h"
#include "ns3/mobility-module.h"
#include "ns3/applications-module.h"
using namespace ns3;

int main(int argc, char* argv[]) {
  uint32_t nCsma = 3, nWifi = 3;

  // ── P2P backbone ──────────────────────
  NodeContainer p2pNodes;
  p2pNodes.Create(2);   // n0(AP), n1

  PointToPointHelper p2p;
  p2p.SetDeviceAttribute(
    "DataRate", StringValue("5Mbps"));
  p2p.SetChannelAttribute(
    "Delay", StringValue("2ms"));
  NetDeviceContainer p2pDevs =
    p2p.Install(p2pNodes);

  // ── CSMA network ─────────────────────
  NodeContainer csmaNodes;
  csmaNodes.Add(p2pNodes.Get(1));
  csmaNodes.Create(nCsma);
  CsmaHelper csma;
  csma.SetChannelAttribute(
    "DataRate", StringValue("100Mbps"));
  csma.SetChannelAttribute(
    "Delay",
    TimeValue(NanoSeconds(6560)));
  NetDeviceContainer csmaDevs =
    csma.Install(csmaNodes);

  // ── WiFi network ──────────────────────
  NodeContainer wifiSta;
  wifiSta.Create(nWifi);
  NodeContainer wifiAp =
    p2pNodes.Get(0);     // n0 is AP

  YansWifiChannelHelper ch =
    YansWifiChannelHelper::Default();
  YansWifiPhyHelper phy =
    YansWifiPhyHelper::Default();
  phy.SetChannel(ch.Create());

  WifiHelper wifi;
  WifiMacHelper mac;
  Ssid ssid = Ssid("ns-3-ssid");

  // STA devices
  mac.SetType("ns3::StaWifiMac",
    "Ssid", SsidValue(ssid),
    "ActiveProbing",
    BooleanValue(false));
  NetDeviceContainer staDevs =
    wifi.Install(phy, mac, wifiSta);

  // AP device
  mac.SetType("ns3::ApWifiMac",
    "Ssid", SsidValue(ssid));
  NetDeviceContainer apDevs =
    wifi.Install(phy, mac, wifiAp);

  // ── Mobility ──────────────────────────
  MobilityHelper mob;
  mob.SetPositionAllocator(
    "ns3::GridPositionAllocator",
    "MinX", DoubleValue(0.0),
    "MinY", DoubleValue(0.0),
    "DeltaX", DoubleValue(5.0),
    "GridWidth",
    UintegerValue(3));
  mob.SetMobilityModel(
    "ns3::RandomWalk2dMobilityModel",
    "Bounds",
    RectangleValue(
      Rectangle(-50,50,-50,50)));
  mob.Install(wifiSta);
  mob.SetMobilityModel(
    "ns3::ConstantPositionMobilityModel");
  mob.Install(wifiAp);

  // ── Internet stack ────────────────────
  InternetStackHelper stack;
  stack.Install(csmaNodes);  // n1 + extra
  stack.Install(wifiAp);    // n0 (AP)
  stack.Install(wifiSta);   // STAs

  // ── IP addresses ─────────────────────
  Ipv4AddressHelper addr;

  addr.SetBase(
    "10.1.1.0", "255.255.255.0");
  Ipv4InterfaceContainer p2pIf =
    addr.Assign(p2pDevs);

  addr.SetBase(
    "10.1.2.0", "255.255.255.0");
  Ipv4InterfaceContainer csmaIf =
    addr.Assign(csmaDevs);

  addr.SetBase(
    "10.1.3.0", "255.255.255.0");
  addr.Assign(staDevs);
  addr.Assign(apDevs);

  // ── Routing ───────────────────────────
  Ipv4GlobalRoutingHelper::
    PopulateRoutingTables();

  // ── Apps: server on CSMA, client on STA
  UdpEchoServerHelper server(9);
  ApplicationContainer sApps =
    server.Install(
      csmaNodes.Get(nCsma));
  sApps.Start(Seconds(1.0));
  sApps.Stop(Seconds(10.0));

  UdpEchoClientHelper client(
    csmaIf.GetAddress(nCsma), 9);
  client.SetAttribute(
    "MaxPackets", UintegerValue(1));
  client.SetAttribute(
    "PacketSize", UintegerValue(1024));
  ApplicationContainer cApps =
    client.Install(
      wifiSta.Get(nWifi-1));
  cApps.Start(Seconds(2.0));
  cApps.Stop(Seconds(10.0));

  // ── MUST STOP — WiFi beacons! ─────────
  Simulator::Stop(Seconds(10.0));

  // ── Tracing ───────────────────────────
  p2p.EnablePcapAll("third");
  phy.EnablePcap(
    "third", apDevs.Get(0));
  csma.EnablePcap(
    "third", csmaDevs.Get(0), true);

  Simulator::Run();
  Simulator::Destroy();
  return 0;
}
06

Dumbbell Topology

Two groups of nodes connected through a bottleneck P2P link. Classic for congestion control testing.

// TOPOLOGY DIAGRAM
n0 ─┐ ┌─ n3 n1 ─┤── R0 ══BOTTLENECK══ R1 ─┤─ n4 n2 ─┘ (1Mbps/10ms) └─ n5 Senders Receivers
Key Points:
  • Used to study congestion control, queue behavior, bottleneck bandwidth
  • Bottleneck link has low bandwidth + high delay — all traffic must cross it
  • Left spokes: fast links (10Mbps). Bottleneck: slow (1Mbps). Right spokes: fast.
  • 6 subnets total: 3 left spokes + 1 bottleneck + 3 right spokes
  • All senders → R0 → bottleneck → R1 → receivers
// DUMBBELL — R0 <--bottleneck--> R1
#include "ns3/core-module.h"
#include "ns3/network-module.h"
#include "ns3/internet-module.h"
#include "ns3/point-to-point-module.h"
#include "ns3/applications-module.h"
using namespace ns3;

int main(int argc, char* argv[]) {

  uint32_t nLeft = 3, nRight = 3;

  // ── Routers ───────────────────────────
  NodeContainer routers;
  routers.Create(2);    // R0, R1

  // ── Left & right nodes ────────────────
  NodeContainer leftNodes, rightNodes;
  leftNodes.Create(nLeft);   // n0,n1,n2
  rightNodes.Create(nRight); // n3,n4,n5

  // ── Bottleneck link (slow!) ───────────
  PointToPointHelper bottleneck;
  bottleneck.SetDeviceAttribute(
    "DataRate", StringValue("1Mbps"));
  bottleneck.SetChannelAttribute(
    "Delay", StringValue("10ms"));
  NetDeviceContainer rDevs =
    bottleneck.Install(routers);

  // ── Spoke links (fast) ────────────────
  PointToPointHelper spoke;
  spoke.SetDeviceAttribute(
    "DataRate", StringValue("10Mbps"));
  spoke.SetChannelAttribute(
    "Delay", StringValue("1ms"));

  // ── Install Internet stack ────────────
  InternetStackHelper stack;
  stack.Install(routers);
  stack.Install(leftNodes);
  stack.Install(rightNodes);

  // ── IP addresses ─────────────────────
  Ipv4AddressHelper addr;
  uint32_t subnet = 1;

  // Bottleneck: 10.1.0.0/24
  addr.SetBase(
    "10.1.0.0", "255.255.255.0");
  addr.Assign(rDevs);

  // Left spokes: 10.1.1.0, 10.1.2.0 ...
  std::vector<Ipv4InterfaceContainer>
    leftIf, rightIf;

  for (uint32_t i = 0; i < nLeft; i++) {
    NodeContainer pair(
      leftNodes.Get(i),
      routers.Get(0));  // → R0
    NetDeviceContainer d =
      spoke.Install(pair);
    std::ostringstream b;
    b << "10.1." << subnet++ << ".0";
    addr.SetBase(
      b.str().c_str(),
      "255.255.255.0");
    leftIf.push_back(addr.Assign(d));
  }

  for (uint32_t i = 0; i < nRight; i++) {
    NodeContainer pair(
      routers.Get(1),  // R1 →
      rightNodes.Get(i));
    NetDeviceContainer d =
      spoke.Install(pair);
    std::ostringstream b;
    b << "10.2." << subnet++ << ".0";
    addr.SetBase(
      b.str().c_str(),
      "255.255.255.0");
    rightIf.push_back(addr.Assign(d));
  }

  // ── Routing ───────────────────────────
  Ipv4GlobalRoutingHelper::
    PopulateRoutingTables();

  // ── Apps: left→right (0→0, 1→1, 2→2)
  for (uint32_t i = 0; i < nRight; i++) {
    UdpEchoServerHelper srv(9);
    ApplicationContainer s =
      srv.Install(rightNodes.Get(i));
    s.Start(Seconds(1.0));
    s.Stop(Seconds(10.0));

    // rightIf[i].GetAddress(1) = right node IP
    UdpEchoClientHelper cli(
      rightIf[i].GetAddress(1), 9);
    cli.SetAttribute(
      "MaxPackets",
      UintegerValue(5));
    cli.SetAttribute(
      "PacketSize",
      UintegerValue(1024));
    ApplicationContainer c =
      cli.Install(leftNodes.Get(i));
    c.Start(Seconds(2.0));
    c.Stop(Seconds(10.0));
  }

  bottleneck.EnablePcapAll(
    "dumbbell");
  Simulator::Run();
  Simulator::Destroy();
  return 0;
}
